-- =============================================================================
-- ZONA G · E-las-cotizaciones.sql
--
-- Córrelo DESPUÉS de la B, la C y la D, en el SQL Editor de Supabase. No
-- borra ni cambia nada de lo que ya existe: agrega una tabla y una función.
-- Se puede volver a correr sin duplicar nada.
--
-- -----------------------------------------------------------------------------
-- QUÉ ES
--
-- El checkout ya no registra pedidos: manda una SOLICITUD DE COTIZACIÓN.
-- A WhatsApp solo se va el carrito, el código postal y el folio. Los datos
-- de la clienta —nombre, teléfono, correo, dirección— se quedan aquí, en
-- `quotes`, y Caty los ve en Panel → Cotizaciones pendientes.
--
-- El folio (ZC-XXXXXXXX) es lo que amarra las dos cosas: llega un WhatsApp
-- que dice «Folio ZC-4F2A91C0» y en el panel está esa misma cotización con
-- todo lo demás.
--
-- -----------------------------------------------------------------------------
-- LO QUE NO HACE, A PROPÓSITO
--
-- · No descuenta existencias. Una cotización todavía no es una venta. Si
--   descontara, cada pregunta que no se concreta dejaría productos
--   «agotados» sin estarlo.
-- · No gasta el código promocional. Lo guarda tal como lo escribió la
--   clienta, para que Caty lo respete si cierra la venta.
-- · No se puede leer desde fuera. Un visitante no tiene ningún permiso sobre
--   la tabla: solo puede llamar a la función, que guarda y regresa el folio.
--   Leer, cambiar y borrar es nada más del equipo.
--
-- Los precios los pone la BASE, no el navegador: se guardan los vigentes al
-- momento de pedir la cotización, igual que hacía el pedido.
-- =============================================================================

-- =============================================================================
-- 1. LA TABLA
-- =============================================================================

create table if not exists public.quotes (
  id               uuid primary key default gen_random_uuid(),
  code             text not null unique,
  idempotency_key  uuid not null unique,
  user_id          uuid,
  status           text not null default 'pendiente'
                   check (status in ('pendiente','atendida','descartada')),
  customer_name    text not null,
  email            text not null,
  phone            text not null,
  address_line     text not null,
  city             text not null,
  state            text not null,
  postal_code      text not null,
  address_notes    text,
  customer_notes   text,
  internal_notes   text,
  -- Lo que pidió, congelado: nombre, variante, cantidad, precio y la
  -- existencia que había en ese momento (para que Caty vea de un vistazo si
  -- algo ya no alcanzaba).
  items            jsonb not null,
  subtotal         numeric(10,2) not null check (subtotal >= 0),
  promo_code       text,
  shipping_method_name text,
  created_at       timestamptz not null default now(),
  updated_at       timestamptz not null default now()
);

create index if not exists quotes_status_created_idx on public.quotes (status, created_at desc);
create index if not exists quotes_phone_created_idx  on public.quotes (phone, created_at desc);

comment on table public.quotes is
  'Solicitudes de cotización del checkout. No descuentan existencias. Las escribe solo request_quote().';

drop trigger if exists quotes_touch on public.quotes;
create trigger quotes_touch before update on public.quotes
  for each row execute function public.touch_updated_at();

alter table public.quotes enable row level security;

-- Nada para `anon`: ni leer ni escribir. Solo la función.
revoke all on public.quotes from anon;
grant select, update, delete on public.quotes to authenticated;
grant all on public.quotes to service_role;

drop policy if exists "Staff reads quotes" on public.quotes;
create policy "Staff reads quotes" on public.quotes
  for select to authenticated using (public.is_staff(auth.uid()));

drop policy if exists "Staff updates quotes" on public.quotes;
create policy "Staff updates quotes" on public.quotes
  for update to authenticated
  using (public.is_staff(auth.uid())) with check (public.is_staff(auth.uid()));

drop policy if exists "Staff deletes quotes" on public.quotes;
create policy "Staff deletes quotes" on public.quotes
  for delete to authenticated using (public.is_staff(auth.uid()));

-- =============================================================================
-- 2. LA FUNCIÓN
--
-- Todos los parámetros que pueden faltar llevan `default null`. Eso NO es un
-- adorno: Supabase busca la función por los nombres de lo que le mandan, y
-- si la clienta no escoge envío ni código, el navegador no los manda. Sin el
-- default, la función «no existe» y la solicitud rebota. Así se rompía el
-- `place_order` original.
-- =============================================================================

create or replace function public.request_quote(
  p_items            jsonb,
  p_customer         jsonb,
  p_idempotency_key  uuid,
  p_promo_code       text default null,
  p_shipping_method_id uuid default null
) returns jsonb
language plpgsql security definer set search_path = public as $$
declare
  existing   record;
  v_email    text := lower(trim(coalesce(p_customer->>'email','')));
  v_phone    text := trim(coalesce(p_customer->>'phone',''));
  n_items    int;
  v_items    jsonb;
  v_subtotal numeric;
  v_ship     text;
  v_code     text;
begin
  if p_idempotency_key is null then raise exception 'Falta la clave de la solicitud'; end if;

  -- El mismo intento dos veces (doble toque, señal que se cae y reintenta)
  -- regresa la MISMA cotización, no una segunda.
  select code, items, subtotal into existing from public.quotes where idempotency_key = p_idempotency_key;
  if found then
    return jsonb_build_object('code', existing.code, 'items', existing.items,
                              'subtotal', existing.subtotal, 'duplicate', true);
  end if;

  if coalesce(length(trim(p_customer->>'name')),0) < 2 then raise exception 'Escribe tu nombre'; end if;
  if v_email !~ '^[^@\s]+@[^@\s]+\.[^@\s]+$' then raise exception 'Correo no válido'; end if;
  if v_phone !~ '^[0-9 +()-]{10,20}$' then raise exception 'Teléfono no válido'; end if;
  if coalesce(length(trim(p_customer->>'address_line')),0) < 5 then raise exception 'Escribe tu dirección'; end if;
  if coalesce(length(trim(p_customer->>'city')),0) < 2 then raise exception 'Escribe tu ciudad'; end if;
  if coalesce(length(trim(p_customer->>'state')),0) < 2 then raise exception 'Escribe tu estado'; end if;
  if coalesce(p_customer->>'postal_code','') !~ '^[0-9]{5}$' then raise exception 'Código postal no válido'; end if;

  -- Freno a quien llene la tabla a lo tonto: con el mismo teléfono, a lo más
  -- cinco solicitudes por hora. Una clienta de verdad no llega a tanto.
  if (select count(*) from public.quotes
       where phone = v_phone and created_at > now() - interval '1 hour') >= 5 then
    raise exception 'Ya recibimos varias solicitudes tuyas. Te contestamos por WhatsApp en un momento.';
  end if;

  if jsonb_typeof(p_items) is distinct from 'array' then raise exception 'Carrito inválido'; end if;

  with carrito as (
    select (e->>'variant_id')::uuid as variant_id, sum((e->>'quantity')::int)::int as quantity
      from jsonb_array_elements(p_items) e group by 1
  )
  select count(*),
         jsonb_agg(jsonb_build_object(
           'product_id', p.id,
           'variant_id', pv.id,
           'name',       p.name,
           'variant',    pv.name,
           'quantity',   c.quantity,
           'unit_price', coalesce(pv.price, p.base_price),
           'line_total', coalesce(pv.price, p.base_price) * c.quantity,
           'stock',      pv.stock
         ) order by p.name, pv.name),
         coalesce(sum(coalesce(pv.price, p.base_price) * c.quantity), 0)
    into n_items, v_items, v_subtotal
    from carrito c
    join public.product_variants pv on pv.id = c.variant_id and pv.is_active
    join public.products p          on p.id = pv.product_id  and p.is_active;

  if (select count(*) from jsonb_array_elements(p_items)) = 0 then raise exception 'Tu carrito está vacío'; end if;
  if n_items = 0 then raise exception 'Los productos de tu carrito ya no están disponibles'; end if;
  if n_items > 50 then raise exception 'Demasiados productos en una sola solicitud'; end if;
  if exists (select 1 from jsonb_array_elements(v_items) i
              where (i->>'quantity')::int < 1 or (i->>'quantity')::int > 20) then
    raise exception 'Cantidad no válida';
  end if;
  -- Si algo del carrito ya no existe, se dice: no se cotiza a medias en
  -- silencio.
  if n_items < (select count(distinct e->>'variant_id') from jsonb_array_elements(p_items) e) then
    raise exception 'Un producto de tu carrito ya no está disponible. Quítalo y vuelve a intentar.';
  end if;

  if p_shipping_method_id is not null then
    select name into v_ship from public.shipping_methods where id = p_shipping_method_id and is_active;
  end if;

  v_code := 'ZC-' || upper(substr(replace(gen_random_uuid()::text,'-',''), 1, 8));

  insert into public.quotes (code, idempotency_key, user_id, customer_name, email, phone,
    address_line, city, state, postal_code, address_notes, customer_notes,
    items, subtotal, promo_code, shipping_method_name)
  values (v_code, p_idempotency_key, auth.uid(), trim(p_customer->>'name'), v_email, v_phone,
    trim(p_customer->>'address_line'), trim(p_customer->>'city'), trim(p_customer->>'state'),
    p_customer->>'postal_code',
    nullif(left(trim(coalesce(p_customer->>'address_notes','')), 200), ''),
    nullif(left(trim(coalesce(p_customer->>'notes','')), 500), ''),
    v_items, v_subtotal,
    nullif(upper(trim(coalesce(p_promo_code,''))), ''),
    v_ship);

  return jsonb_build_object('code', v_code, 'items', v_items, 'subtotal', v_subtotal, 'duplicate', false);
end $$;

revoke all on function public.request_quote(jsonb, jsonb, uuid, text, uuid) from public;
grant execute on function public.request_quote(jsonb, jsonb, uuid, text, uuid) to anon, authenticated;

-- Que Supabase se entere de la función nueva sin esperar.
notify pgrst, 'reload schema';

-- =============================================================================
-- 3. LA COMPROBACIÓN
-- =============================================================================

do $$
declare n int;
begin
  select count(*) into n from information_schema.tables
   where table_schema='public' and table_name='quotes';
  raise notice '%  Está la tabla de cotizaciones', case when n=1 then 'ok  ' else 'MAL ' end;

  select count(*) into n from pg_policies where schemaname='public' and tablename='quotes';
  raise notice '%  Tiene sus % permisos', case when n=3 then 'ok  ' else 'MAL ' end, n;

  select count(*) into n from pg_proc where proname='request_quote';
  raise notice '%  Está la función que las guarda', case when n=1 then 'ok  ' else 'MAL ' end;

  raise notice '%  Un visitante no puede leer la tabla',
    case when not has_table_privilege('anon','public.quotes','select') then 'ok  ' else 'MAL ' end;
end $$;
