import { Link } from "@tanstack/react-router";
import { Minus, Plus, ShoppingBag, X } from "lucide-react";
import { Sheet, SheetContent, SheetDescription, SheetHeader, SheetTitle } from "@/components/ui/sheet";
import { Button } from "@/components/ui/button";
import { useStore, MAX_QTY } from "@/lib/store";
import { formatMXN } from "@/lib/format";

export function QtyControl({
  value,
  max,
  onChange,
  label,
}: {
  value: number;
  max: number;
  onChange: (v: number) => void;
  label: string;
}) {
  const cap = Math.min(max, MAX_QTY);
  return (
    <div className="inline-flex items-center rounded-full border border-border" role="group" aria-label={label}>
      <button
        type="button"
        onClick={() => onChange(value - 1)}
        disabled={value <= 1}
        className="grid h-9 w-9 place-items-center rounded-full transition-colors hover:text-rosa disabled:opacity-40"
        aria-label="Quitar uno"
      >
        <Minus className="h-3.5 w-3.5" />
      </button>
      <span className="w-8 text-center text-sm tabular-nums" aria-live="polite">
        {value}
      </span>
      <button
        type="button"
        onClick={() => onChange(value + 1)}
        disabled={value >= cap}
        className="grid h-9 w-9 place-items-center rounded-full transition-colors hover:text-rosa disabled:opacity-40"
        aria-label="Agregar uno"
      >
        <Plus className="h-3.5 w-3.5" />
      </button>
    </div>
  );
}

export function CartSheet() {
  const { cartOpen, setCartOpen, lines, setQty, remove, subtotal } = useStore();
  return (
    <Sheet open={cartOpen} onOpenChange={setCartOpen}>
      <SheetContent className="flex w-full flex-col border-border bg-popover sm:max-w-md">
        <SheetHeader>
          <SheetTitle className="font-display text-3xl font-medium">Tu selección</SheetTitle>
          <SheetDescription>Los precios finales se confirman al registrar el pedido.</SheetDescription>
        </SheetHeader>
        {lines.length === 0 ? (
          <div className="flex flex-1 flex-col items-center justify-center gap-4 text-center">
            <ShoppingBag className="h-10 w-10 text-cian" aria-hidden />
            <p className="text-muted-foreground">Tu carrito está vacío por ahora.</p>
            <Button asChild variant="outline" onClick={() => setCartOpen(false)}>
              <Link to="/tienda">Explorar la colección</Link>
            </Button>
          </div>
        ) : (
          <>
            <ul className="-mx-2 flex-1 space-y-4 overflow-y-auto px-2 py-4">
              {lines.map((l) => (
                <li key={l.variantId} className="flex gap-4">
                  <div className="h-24 w-20 shrink-0 overflow-hidden rounded-xl bg-muted">
                    {l.image && <img src={l.image} alt="" className="h-full w-full object-cover" />}
                  </div>
                  <div className="flex flex-1 flex-col">
                    <div className="flex items-start justify-between gap-2">
                      <Link
                        to="/producto/$slug"
                        params={{ slug: l.slug }}
                        onClick={() => setCartOpen(false)}
                        className="font-display text-lg leading-tight hover:text-rosa"
                      >
                        {l.name}
                      </Link>
                      <button
                        onClick={() => remove(l.variantId)}
                        className="text-muted-foreground hover:text-foreground"
                        aria-label={`Quitar ${l.name}`}
                      >
                        <X className="h-4 w-4" />
                      </button>
                    </div>
                    {l.variantName && <p className="text-xs text-muted-foreground">{l.variantName}</p>}
                    <div className="mt-auto flex items-center justify-between pt-2">
                      <QtyControl
                        value={l.quantity}
                        max={l.maxStock}
                        onChange={(v) => setQty(l.variantId, v)}
                        label={`Cantidad de ${l.name}`}
                      />
                      <span className="text-sm font-semibold">{formatMXN(l.price * l.quantity)}</span>
                    </div>
                  </div>
                </li>
              ))}
            </ul>
            <div className="space-y-3 border-t border-border pt-4">
              <div className="flex justify-between text-sm">
                <span className="text-muted-foreground">Subtotal estimado</span>
                <span className="font-semibold">{formatMXN(subtotal)}</span>
              </div>
              <p className="text-xs text-muted-foreground">Envío y descuentos se calculan en el siguiente paso.</p>
              <Button asChild size="lg" className="w-full" onClick={() => setCartOpen(false)}>
                <Link to="/checkout">Continuar con mi pedido</Link>
              </Button>
              <Button asChild variant="link" className="w-full" onClick={() => setCartOpen(false)}>
                <Link to="/carrito">Ver carrito completo</Link>
              </Button>
            </div>
          </>
        )}
      </SheetContent>
    </Sheet>
  );
}
