import { Link } from "@tanstack/react-router";
import { Heart, Plus } from "lucide-react";
import { toast } from "sonner";
import { useStore } from "@/lib/store";
import { formatMXN } from "@/lib/format";
import { productFromPrice, productStock, sortVariants, variantPrice, type Product } from "@/lib/catalog";
import { cn } from "@/lib/utils";

export function ProductBadges({ product }: { product: Pick<Product, "badge_new" | "badge_favorite" | "badge_special"> }) {
  const badges = [
    product.badge_new && "Nuevo",
    product.badge_favorite && "Favorito",
    product.badge_special && "Edición especial",
  ].filter(Boolean) as string[];
  if (!badges.length) return null;
  return (
    <div className="flex flex-wrap gap-1.5">
      {badges.map((b) => (
        <span key={b} className="glass rounded-full px-2.5 py-1 text-[0.65rem] font-semibold uppercase tracking-[0.15em] text-champagne">
          {b}
        </span>
      ))}
    </div>
  );
}

export function ProductCard({ product, className }: { product: Product; className?: string }) {
  const { add, setCartOpen, isFavorite, toggleFavorite, hydrated } = useStore();
  const variants = sortVariants(product.product_variants);
  const stock = productStock(product);
  const from = productFromPrice(product);
  const compare = product.compare_at_price ? Number(product.compare_at_price) : null;
  const fav = hydrated && isFavorite(product.id);
  const single = variants.length === 1 ? variants[0] : null;
  const variantSummary = variants.length > 1 ? `${variants.length} opciones` : null;

  function quickAdd() {
    if (!single) return;
    add({
      variantId: single.id,
      productId: product.id,
      slug: product.slug,
      name: product.name,
      variantName: single.name === "Única" ? null : single.name,
      price: variantPrice(product, single),
      image: product.images[0] ?? null,
      quantity: 1,
      maxStock: single.stock,
    });
    toast.success("Agregado a tu carrito", { description: product.name });
    setCartOpen(true);
  }

  return (
    <article className={cn("group relative flex flex-col", className)}>
      <Link
        to="/producto/$slug"
        params={{ slug: product.slug }}
        className="relative block aspect-[4/5] overflow-hidden rounded-2xl bg-muted shadow-card"
      >
        {product.images[0] ? (
          <img
            src={product.images[0]}
            alt={product.name}
            loading="lazy"
            className="h-full w-full object-cover transition-transform duration-700 group-hover:scale-105"
          />
        ) : (
          <div className="grid h-full w-full place-items-center font-display text-3xl text-muted-foreground">Zona G</div>
        )}
        <div className="absolute left-3 top-3">
          <ProductBadges product={product} />
        </div>
        {stock <= 0 && (
          <span className="glass absolute bottom-3 left-3 rounded-full px-3 py-1 text-xs">Agotado</span>
        )}
      </Link>
      <button
        onClick={() => toggleFavorite(product.id)}
        aria-pressed={fav}
        aria-label={fav ? `Quitar ${product.name} de favoritos` : `Guardar ${product.name} en favoritos`}
        className="glass absolute right-3 top-3 grid h-9 w-9 place-items-center rounded-full transition-colors hover:text-rosa"
      >
        <Heart className={cn("h-4 w-4", fav && "fill-rosa text-rosa")} />
      </button>
      <div className="flex items-start justify-between gap-3 pt-4">
        <div className="min-w-0">
          {product.category && <p className="text-[0.68rem] uppercase tracking-[0.2em] text-muted-foreground">{product.category.name}</p>}
          <Link to="/producto/$slug" params={{ slug: product.slug }} className="mt-1 block font-display text-xl leading-tight hover:text-rosa">
            {product.name}
          </Link>
          <p className="mt-1.5 flex items-baseline gap-2 text-sm">
            <span className="font-semibold">
              {variants.length > 1 && "Desde "}
              {formatMXN(from)}
            </span>
            {compare && compare > from && <span className="text-muted-foreground line-through">{formatMXN(compare)}</span>}
            {variantSummary && <span className="text-xs text-muted-foreground">· {variantSummary}</span>}
          </p>
        </div>
        {single && single.stock > 0 ? (
          <button
            onClick={quickAdd}
            className="bg-gradient-brand grid h-10 w-10 shrink-0 place-items-center rounded-full text-primary-foreground shadow-glow transition-transform hover:scale-105"
            aria-label={`Agregar ${product.name} al carrito`}
          >
            <Plus className="h-4 w-4" />
          </button>
        ) : (
          <Link
            to="/producto/$slug"
            params={{ slug: product.slug }}
            className="shrink-0 rounded-full border border-border px-3 py-2 text-xs hover:border-cian hover:text-cian"
          >
            Ver detalles
          </Link>
        )}
      </div>
    </article>
  );
}
