-- Restricciones, triggers y políticas de inventario
SELECT
  'constraint' AS tipo,
  c.conname AS nombre,
  pg_get_constraintdef(c.oid) AS definicion
FROM pg_constraint c
JOIN pg_class t ON t.oid = c.conrelid
JOIN pg_namespace n ON n.oid = t.relnamespace
WHERE n.nspname = 'public'
  AND t.relname IN ('products', 'product_variants', 'quotes')

UNION ALL

SELECT
  'trigger',
  trigger_name,
  action_statement
FROM information_schema.triggers
WHERE event_object_schema = 'public'
  AND event_object_table IN ('products', 'product_variants', 'quotes')

UNION ALL

SELECT
  'policy',
  tablename || '.' || policyname,
  'cmd=' || cmd || '; roles=' || array_to_string(roles, ',')
    || '; using=' || coalesce(qual, '')
    || '; check=' || coalesce(with_check, '')
FROM pg_policies
WHERE schemaname = 'public'
  AND tablename IN ('products', 'product_variants', 'quotes')

ORDER BY 1, 2;

-- 3. Registro permanente de las cantidades descontadas por cotización.
-- Este registro permite restaurar exactamente las variantes vendidas,
-- aunque cambien posteriormente los datos de la cotización.
CREATE TABLE IF NOT EXISTS public.quote_inventory_movements (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  quote_id uuid NOT NULL
    REFERENCES public.quotes(id) ON DELETE CASCADE,
  variant_id uuid NOT NULL
    REFERENCES public.product_variants(id) ON DELETE RESTRICT,
  quantity integer NOT NULL CHECK (quantity > 0),
  created_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE (quote_id, variant_id)
);

CREATE INDEX IF NOT EXISTS quote_inventory_movements_quote_idx
  ON public.quote_inventory_movements (quote_id);

-- Solo las funciones internas podrán consultar o modificar este registro.
ALTER TABLE public.quote_inventory_movements ENABLE ROW LEVEL SECURITY;

REVOKE ALL ON TABLE public.quote_inventory_movements
  FROM PUBLIC, anon, authenticated;

-- 4. Función única para cambiar el estado y controlar el inventario.
CREATE OR REPLACE FUNCTION public.set_quote_status(
  p_quote_id uuid,
  p_status text,
  p_internal_notes text DEFAULT NULL
)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_quote public.quotes%ROWTYPE;
  r record;
  v_stock integer;
BEGIN
  -- Solo el personal autorizado puede realizar estas operaciones.
  IF NOT public.is_staff(auth.uid()) THEN
    RAISE EXCEPTION 'No tienes permiso para modificar cotizaciones';
  END IF;

  IF p_status NOT IN (
    'pendiente', 'vendido', 'cancelada', 'descartada', 'atendida'
  ) THEN
    RAISE EXCEPTION 'Estado de cotización no válido';
  END IF;

  -- Bloquea la cotización para impedir dos operaciones simultáneas.
  SELECT *
    INTO v_quote
    FROM public.quotes
   WHERE id = p_quote_id
   FOR UPDATE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'No se encontró la cotización';
  END IF;

  -- Si el estado no cambia, solo se actualizan las notas.
  IF v_quote.status = p_status THEN
    UPDATE public.quotes
       SET internal_notes = nullif(trim(coalesce(p_internal_notes, '')), '')
     WHERE id = p_quote_id;

    RETURN jsonb_build_object(
      'success', true,
      'status', p_status,
      'inventory_changed', false
    );
  END IF;

  -- ==========================================================
  -- VENTA: únicamente se permite pasar de pendiente a vendido.
  -- ==========================================================
  IF p_status = 'vendido' THEN

    IF v_quote.status <> 'pendiente' THEN
      RAISE EXCEPTION 'Solo una cotización pendiente puede marcarse como vendida';
    END IF;

    IF v_quote.inventory_applied_at IS NOT NULL THEN
      RAISE EXCEPTION 'El inventario de esta cotización ya fue descontado';
    END IF;

    IF jsonb_typeof(v_quote.items) IS DISTINCT FROM 'array'
       OR jsonb_array_length(v_quote.items) = 0 THEN
      RAISE EXCEPTION 'La cotización no contiene productos válidos';
    END IF;

    -- Agrupa por variante por seguridad, por si hubiera IDs repetidos.
    -- Ordenar los IDs ayuda a evitar bloqueos cruzados entre ventas.
    FOR r IN
      SELECT
        (e->>'variant_id')::uuid AS variant_id,
        sum((e->>'quantity')::integer)::integer AS quantity
      FROM jsonb_array_elements(v_quote.items) AS e
      GROUP BY (e->>'variant_id')::uuid
      ORDER BY (e->>'variant_id')::uuid
    LOOP
      IF r.variant_id IS NULL OR r.quantity IS NULL OR r.quantity < 1 THEN
        RAISE EXCEPTION 'La cotización contiene una variante o cantidad inválida';
      END IF;

      -- Bloquea la variante y comprueba la existencia actual.
      SELECT stock
        INTO v_stock
        FROM public.product_variants
       WHERE id = r.variant_id
       FOR UPDATE;

      IF NOT FOUND THEN
        RAISE EXCEPTION 'Una variante de la cotización ya no existe';
      END IF;

      IF v_stock < r.quantity THEN
        RAISE EXCEPTION
          'Existencia insuficiente. Solicitadas: %, disponibles: %',
          r.quantity, v_stock;
      END IF;

      UPDATE public.product_variants
         SET stock = stock - r.quantity
       WHERE id = r.variant_id;

      INSERT INTO public.quote_inventory_movements (
        quote_id, variant_id, quantity
      )
      VALUES (
        p_quote_id, r.variant_id, r.quantity
      );
    END LOOP;

    UPDATE public.quotes
       SET status = 'vendido',
           internal_notes = nullif(trim(coalesce(p_internal_notes, '')), ''),
           inventory_applied_at = now(),
           inventory_reversed_at = NULL
     WHERE id = p_quote_id;

    RETURN jsonb_build_object(
      'success', true,
      'status', 'vendido',
      'inventory_changed', true
    );
  END IF;

  -- ==========================================================
  -- CANCELACIÓN: solo una venta puede cancelarse.
  -- Restaura las cantidades del registro original.
  -- ==========================================================
  IF p_status = 'cancelada' THEN

    IF v_quote.status <> 'vendido' THEN
      RAISE EXCEPTION 'Solo una venta puede marcarse como cancelada';
    END IF;

    IF v_quote.inventory_applied_at IS NULL
       OR v_quote.inventory_reversed_at IS NOT NULL THEN
      RAISE EXCEPTION 'No se puede restaurar el inventario de esta cotización';
    END IF;

    IF NOT EXISTS (
      SELECT 1
        FROM public.quote_inventory_movements
       WHERE quote_id = p_quote_id
    ) THEN
      RAISE EXCEPTION 'No se encontró el registro del inventario descontado';
    END IF;

    FOR r IN
      SELECT variant_id, quantity
        FROM public.quote_inventory_movements
       WHERE quote_id = p_quote_id
       ORDER BY variant_id
    LOOP
      UPDATE public.product_variants
         SET stock = stock + r.quantity
       WHERE id = r.variant_id;

      IF NOT FOUND THEN
        RAISE EXCEPTION 'No se pudo restaurar una variante del inventario';
      END IF;
    END LOOP;

    UPDATE public.quotes
       SET status = 'cancelada',
           internal_notes = nullif(trim(coalesce(p_internal_notes, '')), ''),
           inventory_reversed_at = now()
     WHERE id = p_quote_id;

    RETURN jsonb_build_object(
      'success', true,
      'status', 'cancelada',
      'inventory_changed', true
    );
  END IF;

  -- ==========================================================
  -- DESCARTAR: solo cotizaciones que todavía no son ventas.
  -- ==========================================================
  IF p_status = 'descartada' THEN
    IF v_quote.status <> 'pendiente' THEN
      RAISE EXCEPTION 'Solo una cotización pendiente puede descartarse';
    END IF;

    UPDATE public.quotes
       SET status = 'descartada',
           internal_notes = nullif(trim(coalesce(p_internal_notes, '')), '')
     WHERE id = p_quote_id;

    RETURN jsonb_build_object(
      'success', true,
      'status', 'descartada',
      'inventory_changed', false
    );
  END IF;

  -- No se permite regresar una venta cancelada o descartada a pendiente.
  RAISE EXCEPTION 'No está permitido cambiar de % a %',
    v_quote.status, p_status;
END;
$$;

}

function estadosPermitidos(actual: string): string[] {
  switch (actual) {
    case "pendiente":
      return ["pendiente", "vendido", "descartada"];
    case "vendido":
      return ["vendido", "cancelada"];
    case "cancelada":
      return ["cancelada"];
    case "descartada":
      return ["descartada"];
    case "atendida":
      return ["atendida"];
    default:
      return [actual];
  }
}

          <p className="text-muted-foreground">{quote.phone}</p>

          </a>
        </div>

        })}

        </li>

        {quote.promo_code && (
          <li className="text-muted-foreground">
            Escribió el código {quote.promo_code} (no se ha aplicado ni gastado).
          </li>
        )}
      </ul>

      <div className="grid gap-3">

          <span className="font-medium">Estado</span>
          <select
            value={status}
            onChange={(e) => setStatus(e.target.value)}
            className={selectCls}
            disabled={saving}
          >
            {opcionesEstado.map((k) => (
              <option key={k} value={k}>
                {QUOTE_STATUS_LABEL[k] ?? k}
              </option>
            ))}
          </select>
        </label>

        <AreaField
          id="qnotes"
          label="Notas internas (solo el equipo)"
          rows={2}
          value={notes}
          onChange={(e) => setNotes(e.target.value)}
        />
      </div>

      <Button onClick={save} disabled={saving} className="w-full">
        {saving && <Loader2 className="animate-spin" />}
        Guardar cambios
      </Button>
    </div>
  );
}

Comprueba que muestra los productos, variantes (color/talla), cantidades y precios correctos.

No cambies todavía su estado.

2. Comprueba las opciones
Dime qué sucede al abrir el selector de estado:

¿Aparecen Pendiente, Vendido y Descartada?

¿Puedes seleccionar cada una?

¿Pide confirmación antes de cambiar?

3. Necesito verificar el inventario
En Supabase, consulta las existencias de una variante que aparezca en una cotización. No modifiques nada. Anota su cantidad actual.

4. No ejecutes todavía estas operaciones:

Pendiente → Vendido.

Vendido → Descartada.

Descartada → Vendido.

Primero quiero revisar el código que implementa esas transiciones para evitar que una prueba descuente o devuelva inventario incorrectamente.

Una pregunta importante

¿En la pantalla de cotizaciones puedes abrir el detalle de cada cotización y ver qué productos, colores, tallas y cantidades incluye?

Con eso decidimos el siguiente paso. La prioridad es validar que el inventario se descuente una sola vez al vender y se restaure exactamente una sola vez al cancelar.
